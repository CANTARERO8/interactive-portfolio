WITH RECURSIVE telemetry_hierarchy AS (
  SELECT id, parent_id, node_name, 1 AS depth
  FROM system_nodes WHERE parent_id IS NULL
  UNION ALL
  SELECT n.id, n.parent_id, n.node_name, h.depth + 1
  FROM system_nodes n
  INNER JOIN telemetry_hierarchy h ON n.parent_id = h.id
)
SELECT depth, COUNT(*) FROM telemetry_hierarchy
GROUP BY depth ORDER BY depth ASC;
